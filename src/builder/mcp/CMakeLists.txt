# The mcp module: caliper-mcp, the stdio MCP server that an outside agent
# launches to drive the running Caliper app (docs/plans/mcp-server.md).
#   caliper_builder_mcp   the protocol engine, the loopback HTTP client, the
#                         API file, the tools and the resources
#   caliper-mcp           the executable
find_package(Threads REQUIRED)

# The spec format's JSON Schema is compiled in, so the resource needs no file
# at run time. Configuring again picks up a change to the schema.
get_filename_component(_spec_schema "${CMAKE_CURRENT_SOURCE_DIR}/../spec/schema/caliper-spec-v0.schema.json" ABSOLUTE)
set(SPEC_SCHEMA_BYTES "")
if(EXISTS "${_spec_schema}")
    # configure_file makes the schema a dependency of the configure step.
    configure_file("${_spec_schema}" "${CMAKE_CURRENT_BINARY_DIR}/caliper-spec-v0.schema.json" COPYONLY)
    file(READ "${CMAKE_CURRENT_BINARY_DIR}/caliper-spec-v0.schema.json" _schema_hex HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," SPEC_SCHEMA_BYTES "${_schema_hex}")
endif()
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/spec_schema_embed.cpp" CONTENT
"// Generated by src/builder/mcp/CMakeLists.txt from src/builder/spec/schema/caliper-spec-v0.schema.json.
namespace caliper_builder::mcp::detail {
namespace {
const unsigned char kBytes[] = {@SPEC_SCHEMA_BYTES@ 0x00};
}
extern const char* const kSpecSchemaText;
const char* const kSpecSchemaText = reinterpret_cast<const char*>(kBytes);
}  // namespace caliper_builder::mcp::detail
" @ONLY)

add_library(caliper_builder_mcp STATIC
    codec.cpp
    loopback_http.cpp
    api_file.cpp
    tools.cpp
    resources.cpp
    app_client.cpp
    server.cpp
    "${CMAKE_CURRENT_BINARY_DIR}/spec_schema_embed.cpp")
target_include_directories(caliper_builder_mcp PUBLIC "${CALIPER_BUILDER_INCLUDE_ROOT}")
target_link_libraries(caliper_builder_mcp PUBLIC caliper_builder_commands nlohmann_json::nlohmann_json Threads::Threads)
target_compile_features(caliper_builder_mcp PUBLIC cxx_std_17)
target_compile_definitions(caliper_builder_mcp PRIVATE CALIPER_MCP_VERSION="0.1.0")

add_executable(caliper-mcp main.cpp)
target_link_libraries(caliper-mcp PRIVATE caliper_builder_mcp)

if(BUILD_TESTS)
    set(_mcp_test_sources
        tests/test_codec.cpp
        tests/test_api_file.cpp
        tests/test_tools.cpp
        tests/test_resources.cpp
        tests/test_server.cpp)
    if(NOT WIN32)
        # The loopback client, the fake API and the child process are POSIX
        # only for now.
        list(APPEND _mcp_test_sources tests/fake_http.cpp tests/test_http.cpp tests/test_process.cpp)
    endif()
    caliper_builder_add_tests(mcp SOURCES ${_mcp_test_sources} LIBS caliper_builder_mcp caliper_builder_acp)
    add_dependencies(caliper_builder_mcp_tests caliper-mcp)
    target_compile_definitions(caliper_builder_mcp_tests PRIVATE
        CALIPER_MCP_EXE="$<TARGET_FILE:caliper-mcp>")

    # The acceptance test of the MCP route: caliper-mcp against
    # caliper-builder-host, the session module's stand-in for the app's command
    # endpoints, which exists only when DuckDB is given.
    if(NOT WIN32 AND TARGET caliper_builder_session AND TARGET caliper-builder-host)
        caliper_builder_add_tests(mcp_acceptance SOURCES tests/test_acceptance_mcp.cpp
            LIBS caliper_builder_mcp caliper_builder_session caliper_builder_acp caliper_builder_duckdb)
        add_dependencies(caliper_builder_mcp_acceptance_tests caliper-mcp caliper-builder-host)
        target_compile_definitions(caliper_builder_mcp_acceptance_tests PRIVATE
            CALIPER_MCP_EXE="$<TARGET_FILE:caliper-mcp>"
            CALIPER_BUILDER_HOST_EXE="$<TARGET_FILE:caliper-builder-host>")
    endif()

    # The end-to-end test against the real app, after contracts sections 14
    # and 17. It runs when CALIPER_BUILDER_TEST_APP names a caliper executable,
    # and reports itself skipped otherwise.
    if(NOT WIN32)
        set(CALIPER_BUILDER_TEST_APP "" CACHE FILEPATH
            "A caliper executable for the end-to-end test of caliper-mcp; empty skips it")
        caliper_builder_add_tests(mcp_app SOURCES tests/test_app_e2e.cpp LIBS caliper_builder_mcp caliper_builder_acp)
        add_dependencies(caliper_builder_mcp_app_tests caliper-mcp)
        target_include_directories(caliper_builder_mcp_app_tests SYSTEM PRIVATE
            "${CMAKE_CURRENT_SOURCE_DIR}/../../../third_party/stb")
        target_compile_definitions(caliper_builder_mcp_app_tests PRIVATE
            CALIPER_MCP_EXE="$<TARGET_FILE:caliper-mcp>"
            CALIPER_BUILDER_TEST_APP="${CALIPER_BUILDER_TEST_APP}")
        set_tests_properties(builder_mcp_app PROPERTIES
            SKIP_REGULAR_EXPRESSION "caliper-mcp end-to-end test skipped"
            TIMEOUT 300)
    endif()
endif()
