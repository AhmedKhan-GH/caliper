# caliper_builder_operators: the built-in operators surface and path
# (design/instrument-interface.md, Operators; include/caliper_builder/operators/operators.h):
# their CPU references, their WGSL sources, compiled in, and the program
# declaration of a placed operator.
set(_operator_wgsl surface path)
set(_operator_embed "${CMAKE_CURRENT_BINARY_DIR}/operators_wgsl_embed.cpp")
set(CALIPER_OPERATOR_WGSL_ENTRIES "")
foreach(_op ${_operator_wgsl})
    set(_src "${CMAKE_CURRENT_SOURCE_DIR}/wgsl/${_op}.wgsl")
    # configure_file makes each source a dependency of the configure step.
    configure_file("${_src}" "${CMAKE_CURRENT_BINARY_DIR}/wgsl/${_op}.wgsl" COPYONLY)
    file(READ "${CMAKE_CURRENT_BINARY_DIR}/wgsl/${_op}.wgsl" _hex HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _bytes "${_hex}")
    string(APPEND CALIPER_OPERATOR_WGSL_ENTRIES
        "const unsigned char k_${_op}[] = {${_bytes} 0x00};\n")
endforeach()
file(CONFIGURE OUTPUT "${_operator_embed}" CONTENT
"// Generated by src/builder/operators/CMakeLists.txt from src/builder/operators/wgsl/*.wgsl.
#include <string_view>
namespace caliper_builder::operators::detail {
namespace {
@CALIPER_OPERATOR_WGSL_ENTRIES@}
std::string_view embedded_wgsl(std::string_view op) {
    if (op == \"surface\") return reinterpret_cast<const char*>(k_surface);
    if (op == \"path\") return reinterpret_cast<const char*>(k_path);
    return {};
}
}  // namespace caliper_builder::operators::detail
" @ONLY)

add_library(caliper_builder_operators STATIC operators.cpp "${_operator_embed}")
target_include_directories(caliper_builder_operators PUBLIC "${CALIPER_BUILDER_INCLUDE_ROOT}")
target_link_libraries(caliper_builder_operators PUBLIC caliper_builder_spec nlohmann_json::nlohmann_json)
target_compile_features(caliper_builder_operators PUBLIC cxx_std_17)
set_target_properties(caliper_builder_operators PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(NOT MSVC)
    target_compile_options(caliper_builder_operators PRIVATE -Wall -Wextra)
endif()

caliper_builder_add_tests(operators
    SOURCES
        tests/test_operators.cpp
    LIBS caliper_builder_operators)
